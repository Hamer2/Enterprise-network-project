!S1----------------------------------------------------------------------------

!Default Gateway
ip default-gateway 132.234.27.113

!Trunk Ports

!TRUNK f0/1
int f0/1
switchport mode trunk
switchport trunk native vlan 1000
switchport trunk allowed vlan 20,30,40,99,555,1000

!EtherChannel

!Configure PAgP EtherChannel S1-S2
int range f0/2-3
channel-group 1 mode desirable
exit
int Po1
switchport mode trunk
switchport trunk allowed vlan 20,30,40,99,555,1000
switchport trunk native vlan 1000
exit
int range f0/2-3
no shut

!Configure PAgP EtherChannel S1-S3
int range f0/4-5
channel-group 2 mode desirable
exit
int Po2
switchport mode trunk
switchport trunk allowed vlan 20,30,40,99,555,1000
switchport trunk native vlan 1000
exit
int range f0/4-5
no shut

!Deactivate unused ports and move to Parking_Lot
interface range f0/6-24, g0/1-2
switchport mode access
switchport access vlan 555
shutdown

!SVI Management Config

!Move IP address to VLAN 99
interface vlan 1
no ip address
shutdown
interface vlan 99
ip address 132.234.27.114 255.255.255.240
no shut

!Disable DTP Negotiation
int f0/1
switchport nonegotiate


!Configure S1 as root bridge


spanning-tree vlan 20 priority 4096
spanning-tree vlan 30 priority 4096
spanning-tree vlan 40 priority 4096
spanning-tree vlan 99 priority 4096
spanning-tree vlan 1000 priority 4096
